5/// Describes the input stream's configuration
6#[derive(Copy, Clone, Debug)]
7pub struct InputConfig {
8    /// The number of channels (usually this should be one)
9    pub channel_count: ChannelCount,
10    /// The sample rate the microphone will be recording at
11    pub sample_rate: SampleRate,
12    /// The buffersize, see a thorough explanation in MicrophoneBuilder::with_buffer_size
13    pub buffer_size: cpal::BufferSize,
14    /// The sample format used by the microphone.
15    /// Note we will always convert it to `Sample`
16    pub sample_format: cpal::SampleFormat,
17}
18impl InputConfig {
19    pub(crate) fn supported_given(&self, supported: &cpal::SupportedStreamConfigRange) -> bool {
20        let buffer_ok = match (self.buffer_size, supported.buffer_size()) {
21            (cpal::BufferSize::Default, _) | (_, cpal::SupportedBufferSize::Unknown) => true,
22            (
23                cpal::BufferSize::Fixed(n_frames),
24                cpal::SupportedBufferSize::Range {
25                    min: min_samples,
26                    max: max_samples,
27                },
28            ) => {
29                let n_samples = n_frames * self.channel_count.get() as u32;
30                (*min_samples..*max_samples).contains(&n_samples)
31            }
32        };
33
34        buffer_ok
35            && self.channel_count.get() == supported.channels()
36            && self.sample_format == supported.sample_format()
37            && self.sample_rate.get() <= supported.max_sample_rate()
38            && self.sample_rate.get() >= supported.min_sample_rate()
39    }
40
41    pub(crate) fn with_f32_samples(&self) -> Self {
42        let mut this = *self;
43        this.sample_format = cpal::SampleFormat::F32;
44        this
45    }
46
47    pub(crate) fn stream_config(&self) -> cpal::StreamConfig {
48        cpal::StreamConfig {
49            channels: self.channel_count.get(),
50            sample_rate: self.sample_rate.get(),
51            buffer_size: self.buffer_size,
52        }
53    }
54}
55
56impl From<cpal::SupportedStreamConfig> for InputConfig {
57    fn from(value: cpal::SupportedStreamConfig) -> Self {
58        let buffer_size = match value.buffer_size() {
59            cpal::SupportedBufferSize::Range { .. } => cpal::BufferSize::Default,
60            cpal::SupportedBufferSize::Unknown => cpal::BufferSize::Default,
61        };
62        Self {
63            channel_count: NonZero::new(value.channels())
64                .expect("A supported config never has 0 channels"),
65            sample_rate: NonZero::new(value.sample_rate())
66                .expect("A supported config produces samples"),
67            buffer_size,
68            sample_format: value.sample_format(),
69        }
70    }
71}
72
73impl Default for InputConfig {
74    fn default() -> Self {
75        Self {
76            channel_count: nz!(1),
77            sample_rate: nz!(44_100),
78            buffer_size: cpal::BufferSize::Default,
79            sample_format: cpal::SampleFormat::F32,
80        }
81    }
