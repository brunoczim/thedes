17#[derive(Debug, Error)]
18pub enum InitError {
19    #[error("Failed to create main settings menu")]
20    MainSettingsMenu(#[source] menu::Error),
21    #[error("Failed to create audio settings menu")]
22    AudioSettingsMenu(#[source] menu::Error),

25#[derive(Debug, Error)]
26pub enum LoadError {
27    #[error(transparent)]
28    LoadSettings(#[from] thedes_settings::LoadError),
29    #[error("Failed to initialize component")]
30    Init(#[from] InitError),
31}

33#[derive(Debug, Error)]
34pub enum Error {
35    #[error("Failed to run main settings menu")]
36    MainSettingsMenu(#[source] menu::Error),
37    #[error("Failed to run audio settings menu")]
38    AudioSettingsMenu(#[source] menu::Error),
39    #[error("Failed to set audio output volume")]
40    AudioSetVolume(#[from] SetVolumeError),
41    #[error("Failed to manipulate slidebar")]
42    Slidebar(#[from] slidebar::Error),
43    #[error("Failed to save settings")]
44    Save(#[from] SaveError),
45}

73#[derive(Debug, Clone)]
74pub struct Component {
75    main_settings_menu: Menu<MainSettingsItem, Cancellable>,
76    audio_settings_menu: Menu<AudioSettingsItem, Cancellable>,
77    audio_music_slidebar: Slidebar,
78    settings: Settings,
79    path: PathBuf,
80}
81
82impl Component {
83    pub fn new(path: PathBuf, settings: Settings) -> Result<Self, InitError> {
84        let main_settings_menu = Menu::from_cancellation(
85            "% Settings %",
86            [MainSettingsItem::Audio],
87            Cancellable::new(false),
88        )
89        .map_err(InitError::MainSettingsMenu)?;
90
91        let audio_settings_menu = Menu::from_cancellation(
92            "((|>  Audio Settings  <|))",
93            [AudioSettingsItem::Music],
94            Cancellable::new(false),
95        )
96        .map_err(InitError::AudioSettingsMenu)?;
97
98        let audio_music_slidebar = Slidebar::new(
99            "Set Music Volume",
100            slidebar::Config {
101                ui_size: 17,
102                logical_size: 256,
103                logical_current: settings
104                    .audio()
105                    .volume(AudioSinkType::Music)
106                    .into(),
107            },
108        );
109
110        Ok(Self {
111            path,
112            settings,
113            main_settings_menu,
114            audio_settings_menu,
115            audio_music_slidebar,
116        })
117    }
118
119    pub async fn load(path: PathBuf) -> Result<Self, LoadError> {
120        let settings = Settings::load(&path).await?;
121        Ok(Self::new(path, settings)?)
122    }
123
124    pub async fn save(&self) -> Result<(), SaveError> {
125        self.settings.save(&self.path).await?;
126        Ok(())
127    }
128
129    pub fn values(&self) -> &Settings {
130        &self.settings
131    }
132
133    pub fn values_mut(&mut self) -> &mut Settings {
134        &mut self.settings
135    }
136
137    pub async fn run(&mut self, app: &mut App) -> Result<(), Error> {
138        loop {
139            self.main_settings_menu
140                .run(app)
141                .await
142                .map_err(Error::MainSettingsMenu)?;
143            match self.main_settings_menu.output() {
144                Some(MainSettingsItem::Audio) => loop {
145                    self.audio_settings_menu
146                        .run(app)
147                        .await
148                        .map_err(Error::AudioSettingsMenu)?;
149                    match self.audio_settings_menu.output() {
150                        Some(AudioSettingsItem::Music) => {
151                            self.audio_music_slidebar
152                                .run(app, |app, current| {
153                                    let level = current as Volume;
154                                    self.settings.audio_mut().set_volume(
155                                        AudioSinkType::Music,
156                                        level,
157                                    );
158                                    app.audio_controller.queue([
159                                        audio::Command::new_set_volume(
160                                            AudioSinkType::Music,
161                                            level,
162                                        ),
163                                    ]);
164                                    _ = app.audio_controller.flush();
165                                })
166                                .await?;
167                        },
168                        None => break,
169                    }
170                },
171                None => break,
172            }
173        }
174
175        self.save().await?;
176
177        Ok(())
178    }
179}
