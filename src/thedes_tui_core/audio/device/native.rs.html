16struct NativeAudioDevice {
17    stream: Option<rodio::OutputStream>,
18}
19
20impl AudioDevice for NativeAudioDevice {
21    fn open_sink(&mut self) -> Result<Box<dyn AudioSinkDevice>, OpenSinkError> {
22        let stream = match self.stream.take() {
23            Some(s) => s,
24            None => rodio::OutputStreamBuilder::open_default_stream()
25                .map_err(|e| OpenSinkError::Stream(e.to_string()))?,
26        };
27        let sink = rodio::Sink::connect_new(&stream.mixer());
28        self.stream = Some(stream);
29        Ok(Box::new(NativeAudioSinkDevice { inner: sink }))
30    }
31}
32
33impl fmt::Debug for NativeAudioDevice {
34    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
35        f.debug_struct("NativeAudioDevice")
36            .field("stream", &(&self.stream as *const _))
37            .finish()
38    }
39}
40
41struct NativeAudioSinkDevice {
42    inner: rodio::Sink,
43}
44
45impl fmt::Debug for NativeAudioSinkDevice {
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        f.debug_struct("NativeAudioSink")
48            .field("sink", &(&self.inner as *const _))
49            .finish()
50    }
51}
52
53impl AudioSinkDevice for NativeAudioSinkDevice {
54    fn play_now(&mut self, bytes: &'static [u8]) -> Result<(), PlayNowError> {
55        let reader = Cursor::new(bytes);
56        let source = rodio::Decoder::try_from(reader)
57            .map_err(|e| PlayNowError::Decode(e.to_string()))?;
58
59        self.inner.clear();
60        self.inner.append(source);
61        self.inner.play();
62
63        Ok(())
64    }
65
66    fn set_volume(&mut self, volume: f32) -> Result<(), SetVolumeError> {
67        self.inner.set_volume(volume);
68        Ok(())
69    }
70
71    fn is_playing(&self) -> Result<bool, CheckPlayStatusError> {
72        Ok(!self.inner.empty())
73    }
74}
